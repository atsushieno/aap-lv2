cmake_minimum_required(VERSION 3.18)
project(aap_lv2_state_regression LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(dep "${repo}/external")
set(src "${repo}/androidaudioplugin-lv2/src/main/cpp/src")
set(SANITIZERS "undefined" CACHE STRING "Sanitizers to use (for example address,undefined)")
if(SANITIZERS)
  add_compile_options(-fsanitize=${SANITIZERS} -fno-omit-frame-pointer)
  add_link_options(-fsanitize=${SANITIZERS})
endif()
include_directories("${dep}" "${dep}/lilv/src" "${dep}/sord/src" "${dep}/serd/src" "${dep}/aap-core/include" "${dep}/aap-core/external/cmidi2"
  "${src}" "${dep}/lv2" "${dep}/lilv/include" "${dep}/serd/include"
  "${dep}/sord/include" "${dep}/sratom/include")
set(library_sources)
foreach(component lilv serd sord sratom)
  if(component STREQUAL "lilv")
    set(names plugin pluginclass filesystem ui util state lib zix/tree instance node world query scalepoint collections port)
  elseif(component STREQUAL "serd")
    set(names base64 writer system reader n3 byte_source uri env node string)
  elseif(component STREQUAL "sord")
    set(names syntax sord)
  else()
    set(names sratom)
  endif()
  foreach(name IN LISTS names)
    list(APPEND library_sources "${dep}/${component}/src/${name}.c")
  endforeach()
endforeach()
add_library(state_dependencies STATIC ${library_sources} "${src}/symap.cpp"
  "${src}/zix/ring.c" "${src}/std_workaround.c" "${src}/abstract_io.c")
target_compile_definitions(state_dependencies PRIVATE ZIX_STATIC ZIX_INTERNAL HAVE_MLOCK=1)
if(APPLE)
  # This vendored Serd calls posix_fadvise whenever fileno is enabled.
  target_compile_definitions(state_dependencies PRIVATE SERD_NO_DEFAULT_CONFIG HAVE_POSIX_MEMALIGN)
endif()
add_library(state_fixture MODULE state-fixture.cpp)
set_target_properties(state_fixture PROPERTIES PREFIX "" SUFFIX ".so")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/fixture.lv2")
set_target_properties(state_fixture PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/fixture.lv2")
configure_file(manifest.ttl "${CMAKE_BINARY_DIR}/fixture.lv2/manifest.ttl" COPYONLY)
add_executable(state_regression state-regression.cpp "${src}/android-audio-plugin-lv2-bridge.cpp")
target_link_libraries(state_regression PRIVATE state_dependencies ${CMAKE_DL_LIBS})
enable_testing()
add_test(NAME state_regression COMMAND state_regression "${CMAKE_BINARY_DIR}/fixture.lv2/")

set_tests_properties(state_regression PROPERTIES TIMEOUT 30)
